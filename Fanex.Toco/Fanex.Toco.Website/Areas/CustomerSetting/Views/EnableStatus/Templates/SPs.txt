ALTER PROCEDURE [dbo].[Acc_CustomerSelHRStatus_andy]
	@custid int,
	@uplineid int = 0,
	@roleid int,
	@hasdownline bit = 0 output
AS
/*
Created: 20150608@Andy.Nguyen
Task : #36563 Standardize all enable status products.
DB: DB05.bodb02

Revisions:
    
Param's Explanation:
• @custid: Customer ID
. @uplineid: Upline customer ID
. @roleid: Customer role ID
. @hasdownline: Identify CustID has downline or not
Example: 
	exec [dbo].[Acc_CustomerSelHRStatus_andy] @custid=1176086, @uplineid=1176083, @roleid=2
*/

BEGIN
	DECLARE 
		@srecommend int,
		@mrecommend int,
		@recommend int,
		@downlineid int,
		
		@DisableUplines151_31 BIT,	
		@DisableUplines151_41 BIT,		
		@DisableUplines151_51 BIT,		
		@DisableUplines152_41 BIT,		
		@DisableUplines152_51 BIT,		
		@DisableUplines153_51 BIT,
		
		@Disable151_31 BIT,	
		@Disable151_41 BIT,		
		@Disable151_51 BIT,		
		@Disable152_41 BIT,		
		@Disable152_51 BIT,		
		@Disable153_51 BIT,
		
		@DisableDownlines151_31 BIT,	
		@DisableDownlines151_41 BIT,		
		@DisableDownlines151_51 BIT,		
		@DisableDownlines152_41 BIT,		
		@DisableDownlines152_51 BIT,		
		@DisableDownlines153_51 BIT,
		
		@Username varchar(50)
	
	SELECT 
		@recommend = recommend, 
		@mrecommend = mrecommend, 
		@srecommend = srecommend,
		@Username = username
	FROM custinfo WITH (NOLOCK) WHERE custid = @custid
	
	SELECT @downlineid = 
		(CASE WHEN @roleid = 4 THEN @mrecommend
			WHEN @roleid = 3 THEN @recommend 
			ELSE @custid END)
			
	SELECT @hasdownline = CASE WHEN EXISTS(SELECT top 1 1 FROM custinfo WITH(NOLOCK) WHERE custid = @downlineid) THEN 1 ELSE 0 END
	
	SELECT @DisableUplines151_31 = isnull(DisableHorseRacing,0),
		   @DisableUplines151_41 = isnull(DisableOpenTote,0),
		   @DisableUplines151_51 = isnull(Disable151_AU,0),
		   @DisableUplines152_41 = isnull(Disable152_UK,0),
		   @DisableUplines152_51 = isnull(Disable152_AU,0),
		   @DisableUplines153_51 = isnull(Disable153_AU,0)
	FROM custinfo with (NOLOCK) WHERE custid = @uplineid
	
	SELECT @Disable151_31 = isnull(DisableHorseRacing,0),
		   @Disable151_41 = isnull(DisableOpenTote,0),
		   @Disable151_51 = isnull(Disable151_AU,0),
		   @Disable152_41 = isnull(Disable152_UK,0),
		   @Disable152_51 = isnull(Disable152_AU,0),
		   @Disable153_51 = isnull(Disable153_AU,0)
	FROM custinfo with (NOLOCK) WHERE custid = @custid
	
		--v151
	SELECT top 1 @DisableDownlines151_31 = isnull(DisableHorseRacing,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and DisableHorseRacing = 1
		
	SELECT top 1 @DisableDownlines151_41 = isnull(DisableOpenTote,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and DisableOpenTote = 1
		
	SELECT top 1 @DisableDownlines151_51 = isnull(Disable151_AU,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and Disable151_AU = 1
	--v152
	SELECT top 1 @DisableDownlines152_41 = isnull(Disable152_UK,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and Disable152_UK = 1
		
	SELECT top 1 @DisableDownlines152_51 = isnull(Disable152_AU,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and Disable152_AU = 1
	--v153
	SELECT top 1 @DisableDownlines153_51 = isnull(Disable153_AU,0)
		FROM custinfo with (NOLOCK) 
		WHERE custid = @downlineid and Disable153_AU = 1
			
	DECLARE @tmpTable as TABLE
	(
		Username varchar(50),
		SportTypeId int,
		BetTypeId int,
		DisableUplines bit,
		Disable bit,
		DisableDownlines bit
	)
	
	INSERT INTO @tmpTable	
	SELECT
		@Username,
		151, 
		15131,
		isnull(@DisableUplines151_31,0)	AS DisableUplines,
		isnull(@Disable151_31,0)			AS Disable,
		isnull(@DisableDownlines151_31,0)	AS DisableDownlines
		
	INSERT INTO @tmpTable
	SELECT
		@Username,
		151,
		15141,
		isnull(@DisableUplines151_41,0)	AS DisableUplines,
		isnull(@Disable151_41,0)			AS Disable,
		isnull(@DisableDownlines151_41,0)	AS DisableDownlines
		
	INSERT INTO @tmpTable		
	SELECT
		@Username,
		151, 
		15151,
		isnull(@DisableUplines151_51,0)	AS DisableUplines,
		isnull(@Disable151_51,0)			AS Disable,
		isnull(@DisableDownlines151_51,0)	AS DisableDownlines
		
	INSERT INTO @tmpTable
	SELECT 
		@Username,
		152,
		15241,
		isnull(@DisableUplines152_41,0)	AS DisableUplines,
		isnull(@Disable152_41,0)			AS Disable,
		isnull(@DisableDownlines152_41,0)	AS DisableDownlines
		
	INSERT INTO @tmpTable
	SELECT
		@Username,
		152, 
		15251,
		isnull(@DisableUplines152_51,0)	AS DisableUplines,
		isnull(@Disable152_51,0)			AS Disable,
		isnull(@DisableDownlines152_51,0)	AS DisableDownlines
		
	INSERT INTO @tmpTable
	SELECT 
		@Username,
		153,
		15351,
		isnull(@DisableUplines153_51,0)	AS DisableUplines,
		isnull(@Disable153_51,0)			AS Disable,
		isnull(@DisableDownlines153_51,0)	AS DisableDownlines
		
	SELECT * FROM @tmpTable
	--GRANT EXECUTE ON [dbo].[Acc_CustomerSelHRStatus_andy] TO bodbAdminNet;
	--GRANT EXECUTE ON [dbo].[Acc_CustomerSelHRStatus_andy] TO bodbExtAccNet;
	--GRANT EXECUTE ON [dbo].[Acc_CustomerSelHRStatus_andy] TO bodbDownlineNet;
END
